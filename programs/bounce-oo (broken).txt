let x = 0, y = 0, z = 0, w = 0, h = 0, d = 0
let dx = 1, dy = 1, dz = 1

class MovingBlock {
  block = "chiseled_stone_bricks"

  x = 0
  y = 0
  z = 0

  set(newX: int, newY: int, newZ: int) {
    World.setBlock(this.x, this.y, this.z, "air")
    World.setBlock(newX, newY, newZ, this.block)

    this.x = newX
    this.y = newY
    this.z = newZ
  }

  move(dx: int, dy: int, dz: int) {
    this.set(this.x + dx, this.y + dy, this.z + dz)
  }
}

let block = new MovingBlock()

function tick() {
  block.move(dx, dy, dz)

  if dx > 0 {
    if block.x >= w  dx = -1
  } else {
    if block.x <= 0  dx = 1
  }

  if dy > 0 {
    if block.y >= h  dy = -1
  } else {
    if block.y <= 0  dy = 1
  }

  if dz > 0 {
    if block.z >= d  dz = -1
  } else {
    if block.z <= 0  dz = 1
  }

  #schedule(tick, 0.1)
}

function start(newX: int, newY: int, newZ: int, newW: int, newH: int, newD: int) {
  World.fillBlocks(x - 1, y - 1, z - 1, x + w + 1, y + h + 1, z + d + 1, "air")

  x = newX
  y = newY
  z = newZ
  w = newW
  h = newH
  d = newD

  World.fillBlocks(x - 1, y - 1, z - 1, x + w + 1, y + h + 1, z + d + 1, "light_gray_stained_glass")
  World.fillBlocks(x, y, z, x + w, y + h, z + d, "air")

  block.set(0, 2, 5)
}

start(750, 100, -440, 15, 10, 20)

tick()

#bind("minecraft:start", start)
